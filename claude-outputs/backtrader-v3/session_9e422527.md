#### Pattern 1: Database Connection Manager
```python
# Proposed: core/database.py
from contextlib import contextmanager
import psycopg2
from psycopg2 import pool

class DatabaseManager:
    _pool = None
    
    @classmethod
    def get_pool(cls):
        if cls._pool is None:
            cls._pool = pool.ThreadedConnectionPool(
                minconn=2, maxconn=10,
                dsn=DatabaseConfig.get_connection_string()
            )
        return cls._pool
    
    @classmethod
    @contextmanager
    def get_connection(cls):
        conn = cls.get_pool().getconn()
        try:
            yield conn
            conn.commit()
        except Exception:
            conn.rollback()
            raise
        finally:
            cls.get_pool().putconn(conn)
```

```python
class ValidationChain:

        self.validators = []

            if not result.is_valid:
```

```python
from abc import ABC, abstractmethod
from dataclasses import dataclass
from typing import Dict, List, Optional
import backtrader as bt

@dataclass
class Signal:
    """Trading signal output"""
    ticker: str
    action: str  # 'BUY', 'SELL', 'HOLD'
    weight: float
    price: float
    quantity: int
    reason: str
    
@dataclass
class StrategyContext:
    """Context passed to strategy methods"""
    date: date
    account_id: str
    positions: Dict[str, float]
    cash: float
    nav: float
    universe: List[str]
    price_data: pd.DataFrame

class BaseStrategy(bt.Strategy, ABC):
    """
    Abstract base class for all strategies.
    Strategies only need to implement business logic methods.
    """
    
    # Required: Strategy metadata
    NAME: str = "base"
    VERSION: str = "1.0.0"
    DESCRIPTION: str = ""
    
    # Injected by framework
    _trade_service: TradeService
    _position_service: PositionService
    _cash_service: CashService
    
    @abstractmethod
    def calculate_signals(self, context: StrategyContext) -> List[Signal]:
        """
        Core strategy logic: calculate trading signals.
        This is the ONLY method most strategies need to implement.
        """
        pass
    
    def calculate_indicators(self, data: bt.DataBase) -> Dict[str, float]:
        """
        Optional: Override to add custom indicators.
        Default implementation provides common indicators.
        """
        return {}
    
    def should_rebalance(self, context: StrategyContext) -> bool:
        """
        Optional: Override to customize rebalance timing.
        Default: every N weeks based on config.
        """
        return self._default_rebalance_check(context)
    
    def filter_universe(self, context: StrategyContext) -> List[str]:
        """
        Optional: Override to filter tradeable universe.
        Default: use config-defined basket.
        """
        return context.universe
    
    # ============ Framework methods (DO NOT override) ============
    
    def __init__(self):
        """Initialize strategy with injected services."""
        super().__init__()
        self._setup_services()
        self._setup_indicators()
        self.on_init()  # Hook for strategy-specific init
    
    def next(self):
        """Main processing loop - delegates to strategy logic."""
        context = self._build_context()
        
        # Process pending trades
        self._trade_service.process_pending(context)
        
        # Check rebalance
        if self.should_rebalance(context):
            signals = self.calculate_signals(context)
            self._execute_signals(signals, context)
    
    def on_init(self):
        """Optional hook for strategy initialization."""
        pass
    
    def on_stop(self):
        """Optional hook for strategy cleanup."""
        pass
```

### 4.2 Strategy Registry (Auto-discovery)

```python
# framework/strategy_registry.py
import importlib
import pkgutil
from pathlib import Path
from typing import Dict, Type

class StrategyRegistry:
    """Auto-discovers and registers strategy plugins."""
    
    _strategies: Dict[str, Type[BaseStrategy]] = {}
    
    @classmethod
    def discover(cls, strategies_path: str = "strategies"):
        """Auto-discover all strategies in the strategies folder."""
        strategies_dir = Path(strategies_path)
        
        for item in strategies_dir.iterdir():
            if item.is_dir() and (item / "strategy.py").exists():
                module_name = f"strategies.{item.name}.strategy"
                module = importlib.import_module(module_name)
                
                # Find BaseStrategy subclass in module
                for attr_name in dir(module):
                    attr = getattr(module, attr_name)
                    if (isinstance(attr, type) and 
                        issubclass(attr, BaseStrategy) and 
                        attr is not BaseStrategy):
                        cls._strategies[attr.NAME] = attr
                        print(f"Registered strategy: {attr.NAME}")
    
    @classmethod
    def get(cls, name: str) -> Type[BaseStrategy]:
        """Get strategy class by name."""
        if name not in cls._strategies:
            raise ValueError(f"Strategy '{name}' not found. "
                           f"Available: {list(cls._strategies.keys())}")
        return cls._strategies[name]
    
    @classmethod
    def list(cls) -> List[str]:
        """List all available strategies."""
        return list(cls._strategies.keys())
```

### 4.4 Example: Implementing New Strategy

```python
# strategies/momentum_etf/strategy.py
from framework.base_strategy import BaseStrategy, StrategyContext, Signal
from framework.mixins import IndicatorsMixin
from typing import List, Dict
import numpy as np

class MomentumETFStrategy(BaseStrategy, IndicatorsMixin):
    """
    ETF Momentum Strategy
    - Ranks stocks by ROC (Rate of Change)
    - Buys top N% performers
    - Equal-weight allocation
    """
    
    NAME = "momentum_etf"
    VERSION = "2.0.0"
    DESCRIPTION = "ETF Momentum with ROC ranking"
    
    def on_init(self):
        """Initialize momentum-specific indicators."""
        for d in self.datas:
            self.setup_momentum_indicators(d)
    
    def calculate_signals(self, context: StrategyContext) -> List[Signal]:
        """
        Core strategy logic: rank by momentum, select top N%.
        This is the ONLY method we need to implement!
        """
        # Calculate momentum scores
        scores = {}
        for ticker in context.universe:
            data = self.getdatabyname(ticker)
            if data is None:
                continue
            
            roc = self.indicators[data]['roc_long'][0]
            if not np.isnan(roc):
                scores[ticker] = roc
        
        # Rank and select top N%
        top_n = int(len(scores) * self.p.top_n_percent)
        ranked = sorted(scores.items(), key=lambda x: x[1], reverse=True)
        selected = [t for t, _ in ranked[:top_n]]
        
        # Generate signals
        signals = []
        weight = 1.0 / len(selected) if selected else 0
        
        for ticker in selected:
            current_weight = self._get_current_weight(ticker, context)
            if abs(weight - current_weight) > self.p.weight_band:
                signals.append(Signal(
                    ticker=ticker,
                    action='BUY' if weight > current_weight else 'SELL',
                    weight=weight,
                    price=context.price_data.loc[ticker, 'close'],
                    quantity=0,  # Will be calculated by framework
                    reason=f"Momentum rank top {self.p.top_n_percent*100}%"
                ))
        
        # Sell positions not in selection
        for ticker in context.positions:
            if ticker not in selected:
                signals.append(Signal(
                    ticker=ticker,
                    action='SELL',
                    weight=0,
                    price=context.price_data.loc[ticker, 'close'],
                    quantity=context.positions[ticker],
                    reason="Not in momentum selection"
                ))
        
        return signals
```

### 4.5 Unified Entry Point

```python
# run.py
import argparse
import yaml
from framework.strategy_registry import StrategyRegistry
from framework.strategy_runner import StrategyRunner
from core.database import DatabaseManager

def main():
    parser = argparse.ArgumentParser(description='BackTrader Strategy Runner')
    parser.add_argument('--strategy', '-s', required=True, 
                       help='Strategy name (e.g., momentum_etf)')
    parser.add_argument('--config', '-c', default=None,
                       help='Custom config file (optional)')
    parser.add_argument('--mode', '-m', default='backtest',
                       choices=['backtest', 'live', 'paper'],
                       help='Running mode')
    parser.add_argument('--from-date', type=str, default=None,
                       help='Start date (YYYY-MM-DD)')
    parser.add_argument('--to-date', type=str, default=None,
                       help='End date (YYYY-MM-DD)')
    parser.add_argument('--cleanup', action='store_true',
                       help='Cleanup database before run')
    
    args = parser.parse_args()
    
    # Initialize database pool
    DatabaseManager.initialize()
    
    # Discover strategies
    StrategyRegistry.discover()
    
    # Get strategy class
    strategy_class = StrategyRegistry.get(args.strategy)
    
    # Load config
    config_path = args.config or f"strategies/{args.strategy}/config.yaml"
    with open(config_path) as f:
        config = yaml.safe_load(f)
    
    # Run strategy
    runner = StrategyRunner(
        strategy_class=strategy_class,
        config=config,
        mode=args.mode
    )
    
    if args.cleanup:
        runner.cleanup_database()
    
    results = runner.run(
        from_date=args.from_date,
        to_date=args.to_date
    )
    
    print(f"Strategy completed. Final NAV: {results.final_nav:,.0f}")

if __name__ == "__main__":
    main()
```
