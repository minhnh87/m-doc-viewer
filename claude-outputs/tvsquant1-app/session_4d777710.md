     def get_csv_mtime(csv_path: Path) -> float:
         """Get CSV file modification timestamp as float (seconds since epoch)."""
         return csv_path.stat().st_mtime

     def get_bundle_mtime(bundle_name: str) -> float | None:
         """Get most recent bundle ingestion timestamp."""
         from datetime import datetime
         bundle_path = get_bundle_path(bundle_name)
         if not bundle_path.exists():
             return None
         ingestion_dirs = [d for d in bundle_path.iterdir() if d.is_dir() and "T" in d.name]
         if not ingestion_dirs:
             return None
         latest = max(ingestion_dirs, key=lambda d: d.name)
         try:
             ts_str = latest.name.replace("T", " ").split(".")[0]
             dt = datetime.strptime(ts_str, "%Y-%m-%d %H:%M:%S")
             return dt.timestamp()
         except ValueError:
             return latest.stat().st_mtime

     def csv_modified_after_bundle(csv_path: Path, bundle_name: str) -> bool:
         """Check if CSV file is newer than the bundle."""
         bundle_mtime = get_bundle_mtime(bundle_name)
         if bundle_mtime is None:
             return True
         csv_mtime = get_csv_mtime(csv_path)
         return csv_mtime > bundle_mtime

     def ensure_bundle_fresh(symbol: str, csv_path: Path, bundle_name: str = DEFAULT_BUNDLE_NAME) -> str:
         """Ensure bundle for a single symbol is up-to-date."""
         if not csv_modified_after_bundle(csv_path, bundle_name):
             return "up-to-date"
         was_existing = bundle_exists(bundle_name)
         if was_existing:
             delete_bundle(bundle_name)
         df = ingest_csv(csv_path)
         create_bundle_from_dataframe(df, symbol, bundle_name=bundle_name)
         return "rebuilt" if was_existing else "created"

     def refresh_bundles(data_dir: str | Path, bundle_prefix: str = "tvsquant_") -> dict[str, str]:
         """Check and refresh bundles for all CSVs in a directory."""
         data_path = Path(data_dir)
         results: dict[str, str] = {}
         csv_files = list(data_path.glob("*.csv"))
         for csv_path in csv_files:
             symbol = csv_path.stem
             bundle_name = f"{bundle_prefix}{symbol.lower()}"
             try:
                 status = ensure_bundle_fresh(symbol, csv_path, bundle_name)
                 results[symbol] = status
                 logger.info(f"Checking {symbol}: {status}")
             except Exception as e:
                 logger.warning(f"Failed to refresh {symbol}: {e}")
                 results[symbol] = "error"
         created = sum(1 for s in results.values() if s == "created")
         rebuilt = sum(1 for s in results.values() if s == "rebuilt")
         skipped = sum(1 for s in results.values() if s == "up-to-date")
         logger.info(f"Refresh complete: {created} created, {rebuilt} rebuilt, {skipped} up-to-date")
         return results
