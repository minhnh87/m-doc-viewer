### Ví dụ 2: Custom Slippage Model (sàn VN)

```python
from zipline.finance.slippage import SlippageModel

class VNSlippage(SlippageModel):
    """Slippage model cho thị trường VN - thanh khoản thấp hơn"""
    
    def __init__(self, volume_limit=0.1, price_impact=0.05):
        self.volume_limit = volume_limit  # Max 10% volume/bar
        self.price_impact = price_impact  # 5% price impact
    
    def process_order(self, data, order):
        volume = data.current(order.asset, 'volume')
        price = data.current(order.asset, 'close')
        
        # Giới hạn khối lượng khớp
        max_fill = int(volume * self.volume_limit)
        fill_amount = min(abs(order.amount), max_fill)
        
        # Tính slippage dựa trên % volume
        volume_share = fill_amount / volume
        slippage = price * self.price_impact * volume_share
        
        fill_price = price + slippage if order.amount > 0 else price - slippage
        
        return fill_price, fill_amount
```

```python
from zipline.finance.commission import CommissionModel

class VNCommission(CommissionModel):
    """Phí giao dịch sàn VN: 0.15% mua + 0.15% bán + 0.1% thuế bán"""
    
    def calculate(self, order, transaction):
        value = abs(transaction.amount * transaction.price)
        
        commission = value * 0.0015  # 0.15% phí MG
        
        if transaction.amount < 0:  # Bán
            commission += value * 0.001  # 0.1% thuế
        
        return commission
```

```python
from zipline.finance.asset_restrictions import Restrictions

class VNTradingRestrictions(Restrictions):
    """Hạn chế giao dịch theo quy định VN"""
    
    def __init__(self):
        self.suspended_stocks = set()  # Cổ phiếu bị đình chỉ
        self.t3_positions = {}         # T+3 settlement
    
    def is_restricted(self, asset, dt):
        # Cổ phiếu đình chỉ
        if asset.symbol in self.suspended_stocks:
            return True
        
        # Chưa đủ T+3 để bán
        if asset in self.t3_positions:
            buy_date = self.t3_positions[asset]
            if (dt - buy_date).days < 3:
                return True  # Không được bán
        
        return False
```

## Cách 2: Custom Pipeline Factor (Nâng cao, khuyến nghị)

```python
from zipline.pipeline import Pipeline, CustomFactor
from zipline.pipeline.factors import AverageDollarVolume

class DaysToExDate(CustomFactor):
    """Factor tính số ngày đến ex-date gần nhất"""
    inputs = []
    window_length = 1
    
    def __init__(self, events_df):
        # events_df: DataFrame với columns [symbol, ex_date, event_type]
        self.events_df = events_df
    
    def compute(self, today, assets, out):
        for i, asset in enumerate(assets):
            symbol = asset.symbol
            events = self.events_df[self.events_df['symbol'] == symbol]
            
            upcoming = events[events['ex_date'] > today]
            if len(upcoming) > 0:
                next_ex = upcoming['ex_date'].min()
                out[i] = (next_ex - today).days
            else:
                out[i] = 999  # Không có sự kiện

class ExDateExitSignal(CustomFactor):
    """Signal = True nếu cần thoát trước ex-date"""
    inputs = [DaysToExDate]
    window_length = 1
    params = {'exit_days': 2}
    
    def compute(self, today, assets, out, days_to_ex):
        out[:] = (days_to_ex > 0) & (days_to_ex <= self.params['exit_days'])
```
